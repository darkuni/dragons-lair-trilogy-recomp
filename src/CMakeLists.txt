cmake_minimum_required(VERSION 3.16)
project(dltrilogy C CXX)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(GEN_DIR gen CACHE STRING "generated code folder (gen_trace for a fully traced debug build)")
file(GLOB GEN_SRC ${GEN_DIR}/*.cpp)
file(GLOB RT_SRC runtime/*.cpp)
# Space Ace red ENERGIZE restoration (same code as the dltfix disc patcher; must stay bit-exact)
file(GLOB RED_SRC redfix/*.c)
set_source_files_properties(${RED_SRC} PROPERTIES COMPILE_OPTIONS "-std=gnu11;-O2;-ffp-contract=off;-fno-fast-math;-w")
add_executable(dltrilogy ${RT_SRC} ${GEN_SRC} ${RED_SRC})
target_include_directories(dltrilogy PRIVATE runtime ${GEN_DIR} .)
if(WIN32)
  # SDL2 development package (SDL2-devel-*-mingw) location
  set(SDL2_DIR "" CACHE PATH "SDL2 mingw dev package root (x86_64-w64-mingw32 subfolder)")
  target_include_directories(dltrilogy PRIVATE runtime/gl ${SDL2_DIR}/include)
  target_compile_definitions(dltrilogy PRIVATE SDL_MAIN_HANDLED)
  target_compile_options(dltrilogy PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_directories(dltrilogy PRIVATE ${SDL2_DIR}/lib)
  target_sources(dltrilogy PRIVATE runtime/app.rc)
  target_link_libraries(dltrilogy PRIVATE SDL2 winmm -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
elseif(USE_SDL)
  target_compile_definitions(dltrilogy PRIVATE USE_SDL)
  target_compile_options(dltrilogy PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_libraries(dltrilogy PRIVATE SDL2 GL pthread -static-libstdc++ -static-libgcc)
else()
  target_compile_options(dltrilogy PRIVATE -O2 -fno-strict-aliasing -w -g1)
  target_link_libraries(dltrilogy PRIVATE EGL GL pthread)
endif()
